# go2ascii Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Go reimplementation of the `convert2ascii` Ruby gem — two CLI binaries `image2ascii` and `video2ascii` with CLI-parameter compatibility — in a new `go2ascii/` directory, using pure-Go image decoding, cgo-bound FFmpeg for video/audio, goroutine-parallel frame conversion, and full-Go audio playback.

**Architecture:** Single Go module (`github.com/Mark24Code/convert2ascii/go2ascii`) exposing a root library package `convert2ascii` (mirrors Ruby's `Convert2Ascii` classes) plus `cmd/image2ascii` and `cmd/video2ascii` binaries. Frame conversion runs through a goroutine worker pool. Terminal playback uses beep/oto for audio and the ANSI alternate screen buffer, with audio-position-based A/V sync.

**Tech Stack:** Go 1.22+ (environment has 1.26), cgo (libavformat/libavcodec/libavutil/libswscale/libswresample), golang.org/x/image (webp + draw), golang.org/x/term, gopxl/beep v2 + hajimehoshi/oto v2.

**Reference spec:** `docs/superpowers/specs/2026-08-21-go2ascii-design.md`
**Ruby reference (read, don't copy):** `exe/`, `lib/convert2ascii/` in the repo root.

**Notation & conventions:**
- All paths in tasks are relative to `/Users/bilibili/Labspace/convert2ascii/go2ascii/` unless prefixed with `repo:` (meaning repo root `/Users/bilibili/Labspace/convert2ascii/`).
- Commands run from inside `go2ascii/` unless noted.
- **Deviation from spec §4.6/§5.6:** idiomatic Go `Generate()` returns `error` instead of fluent self-return. Chaining is not idiomatic in Go; error propagation matters. CLI behavior is unaffected.
- **Deviation from spec §4.2:** transparent pixels composite onto white before rendering (ImageMagick/ASCII convention), so a fully-transparent pixel renders as the brightest char.

**Build prerequisites (already verified on this machine):** Go 1.26.5, pkg-config, FFmpeg 8.x dev libs (libavformat 63.1, libavcodec 63.1, libavutil 61.1, libswscale 10.1, libswresample 7.1). FFmpeg ≥ 6 required (uses `swr_alloc_set_opts2` + `AVChannelLayout`). `CGO_ENABLED=1` required.

The Ruby behavior this replicates (`lib/convert2ascii/image2ascii.rb`):
- scale to `width`, then height = `round(origRows*width/origCols)/2` (integer division)
- per-pixel luminance `0.2126r + 0.7152g + 0.0722b`, RGB in 0–255
- `idx = floor(brightness / (255.0/len(chars)))`, clamped to `len-1`
- newline after every row (`col % (width-1) == 0 && col != 0` fires at last column of each row because scaled columns == width)
- color style: `\x1b[38;2;r;g;bm<char>\x1b[0m`; color_block: `\x1b[48;2;r;g;bm \x1b[0m`

```go
package imagelib

import (
	"bytes"
	"image"
	"image/color"
	_ "image/gif"
	_ "image/jpeg"
	_ "image/png"
	"io"
	"math"
	"net/http"
	"os"
	"strconv"
	"strings"

	_ "golang.org/x/image/bmp" // stdlib has no image/bmp; BMP lives in x/image
	"golang.org/x/image/draw"
	_ "golang.org/x/image/webp"
)

// DefaultChars is the exact character ramp used by the Ruby version.
const DefaultChars = ".'`^\",:;Il!i><~+_-?][}{1)(|\\/tfjrxnuvczXYUJCLQ0OZmwqpdbkhao*#MW&8%B@$"

// Style mirrors the Ruby STYLE_ENUM: "color" (ANSI) or "text" (plain).
type Style string

const (
	StyleColor Style = "color"
	StyleText  Style = "text"
)

// ColorMode mirrors the Ruby COLOR_ENUM: "full" or "greyscale".
type ColorMode string

const (
	ColorFull ColorMode = "full"
	ColorGrey ColorMode = "greyscale"
)

// Options configure a render.
type Options struct {
	Width      int
	Style      Style
	Color      ColorMode
	ColorBlock bool
	Chars      string
}

// Open loads an image from a local path or an http(s) URL.
// GIFs render as their first frame.
func Open(uri string) (image.Image, error) {
	var data []byte
	if strings.HasPrefix(uri, "http://") || strings.HasPrefix(uri, "https://") {
		resp, err := http.Get(uri)
		if err != nil {
			return nil, err
		}
		defer resp.Body.Close()
		if resp.StatusCode != http.StatusOK {
			return nil, io.ErrUnexpectedEOF
		}
		data, err = io.ReadAll(resp.Body)
		if err != nil {
			return nil, err
		}
	} else {
		var err error
		data, err = os.ReadFile(uri)
		if err != nil {
			return nil, err
		}
	}
	img, _, err := image.Decode(bytes.NewReader(data))
	if err != nil {
		return nil, err
	}
	return img, nil
}

// AspectSize computes the output dimensions exactly as the Ruby version's
// two-step scale: width becomes width; height is round(origRows*width/origCols)/2.
func AspectSize(orig image.Image, width int) (w, h int) {
	ocol := orig.Bounds().Dx()
	orow := orig.Bounds().Dy()
	if ocol <= 0 {
		ocol = 1
	}
	scaledRows := int(math.Round(float64(orow) * float64(width) / float64(ocol)))
	return width, scaledRows / 2
}

// Render converts img to an ASCII-art string.
func Render(img image.Image, o Options) string {
	w, h := AspectSize(img, o.Width)
	if w < 2 {
		w = 2
	}
	if h < 1 {
		h = 1
	}

	// Composite onto white so transparent pixels render as bright.
	dst := image.NewRGBA(image.Rect(0, 0, w, h))
	draw.Draw(dst, dst.Bounds(), &image.Uniform{C: color.White}, image.Point{}, draw.Src)
	draw.CatmullRom.Scale(dst, dst.Bounds(), img, img.Bounds(), draw.Over, nil)

	chars := o.Chars
	if chars == "" {
		chars = DefaultChars
	}
	charLen := len(chars)

	style := o.Style
	if style == "" {
		style = StyleColor
	}
	colorMode := o.Color
	if colorMode == "" {
		colorMode = ColorFull
	}

	var sb strings.Builder
	sb.Grow(w*h + h)
	for y := 0; y < h; y++ {
		row := dst.Pix[y*dst.Stride : y*dst.Stride+w*4]
		for x := 0; x < w; x++ {
			i := x * 4
			r := float64(row[i])
			g := float64(row[i+1])
			b := float64(row[i+2])
			brightness := 0.2126*r + 0.7152*g + 0.0722*b
			idx := int(brightness / (255.0 / float64(charLen)))
			if idx >= charLen {
				idx = charLen - 1
			}
			if idx < 0 {
				idx = 0
			}
			ch := chars[idx]

			switch style {
			case StyleText:
				sb.WriteByte(ch)
			default: // StyleColor
				cr, cg, cb := int(r), int(g), int(b)
				if colorMode == ColorGrey {
					cg, cb = cr, cr // avoid go vet self-assignment warning
				}
				if o.ColorBlock {
					sb.WriteString("\x1b[48;2;")
					sb.WriteString(strconv.Itoa(cr))
					sb.WriteString(";")
					sb.WriteString(strconv.Itoa(cg))
					sb.WriteString(";")
					sb.WriteString(strconv.Itoa(cb))
					sb.WriteString("m \x1b[0m")
				} else {
					sb.WriteString("\x1b[38;2;")
					sb.WriteString(strconv.Itoa(cr))
					sb.WriteString(";")
					sb.WriteString(strconv.Itoa(cg))
					sb.WriteString(";")
					sb.WriteString(strconv.Itoa(cb))
					sb.WriteString("m")
					sb.WriteByte(ch)
					sb.WriteString("\x1b[0m")
				}
			}
		}
		sb.WriteByte('\n')
	}
	return sb.String()
}
```

// Package ansi wraps the ANSI escape codes used by the Ruby Terminal helper.
package ansi

import (
	"fmt"
	"os"

	"golang.org/x/term"
)

const (
	clearBuffer = "\x1b[3J"
	clearScreen = "\x1b[2J"
	hideCursor  = "\x1b[?25l"
	showCursor  = "\x1b[?25h"
	openBuffer  = "\x1b[?1049h"
	closeBuffer = "\x1b[?1049l"
)

// ClearBuffer clears the scrollback buffer.
func ClearBuffer() { fmt.Print(clearBuffer) }

// ClearScreen clears the screen.
func ClearScreen() { fmt.Print(clearScreen) }

// HideCursor hides the terminal cursor.
func HideCursor() { fmt.Print(hideCursor) }

// ShowCursor shows the terminal cursor.
func ShowCursor() { fmt.Print(showCursor) }

// OpenBuffer switches to the alternate screen buffer.
func OpenBuffer() { fmt.Print(openBuffer) }

// CloseBuffer leaves the alternate screen buffer.
func CloseBuffer() { fmt.Print(closeBuffer) }

// Size returns the terminal size (rows, cols), falling back to (24, 80).
func Size() (rows, cols int) {
	r, c, err := term.GetSize(int(os.Stdout.Fd()))
	if err != nil || r <= 0 || c <= 0 {
		return 24, 80
	}
	return r, c
}
```

// Package convert2ascii is the public library mirroring the Ruby gem's
// Convert2Ascii::Image2Ascii and Convert2Ascii::Video2Ascii classes.
package convert2ascii

import (
	"fmt"

	"github.com/Mark24Code/convert2ascii/go2ascii/internal/imagelib"
	"github.com/Mark24Code/convert2ascii/go2ascii/internal/ansi"
)

// ImageOptions mirrors the Ruby Image2Ascii constructor/generate kwargs.
type ImageOptions struct {
	URI        string
	Width      int
	Style      string // "color" | "text"; empty -> "color"
	Color      string // "full" | "greyscale"; empty -> "full"
	ColorBlock bool
	Chars      string // empty -> DefaultChars
}

// Image2Ascii converts an image to an ASCII-art string.
type Image2Ascii struct {
	URI        string
	Width      int
	Style      string
	Color      string
	ColorBlock bool
	Chars      string

	asciiString string
}

// NewImage2Ascii builds an Image2Ascii. Width 0 means terminal columns.
func NewImage2Ascii(opts ImageOptions) *Image2Ascii {
	width := opts.Width
	if width <= 0 {
		_, cols := ansi.Size()
		width = cols
	}
	return &Image2Ascii{
		URI:        opts.URI,
		Width:      width,
		Style:      opts.Style,
		Color:      opts.Color,
		ColorBlock: opts.ColorBlock,
		Chars:      opts.Chars,
	}
}

// Generate renders the image. It mutates and returns self on success.
func (a *Image2Ascii) Generate() error {
	img, err := imagelib.Open(a.URI)
	if err != nil {
		return fmt.Errorf("open image: %w", err)
	}
	style := imagelib.Style(a.Style)
	if style == "" {
		style = imagelib.StyleColor
	}
	a.asciiString = imagelib.Render(img, imagelib.Options{
		Width:      a.Width,
		Style:      style,
		Color:      imagelib.ColorMode(a.Color),
		ColorBlock: a.ColorBlock,
		Chars:      a.Chars,
	})
	return nil
}

// String returns the rendered ASCII string.
func (a *Image2Ascii) String() string { return a.asciiString }
```

```go
// Package cli is a small table-driven argument parser that reproduces the
// Ruby OptionParser syntax used by the original executables: -i X, -iX,
// --image X, --image=X, and boolean flags.
package cli

import (
	"fmt"
	"strconv"
	"strings"
)

// Kind of an option.
type Kind int

const (
	// KindBool is a presence flag (no value).
	KindBool Kind = iota
	// KindInt parses an integer.
	KindInt
	// KindString is a plain string value.
	KindString
)

// Option describes one flag.
type Option struct {
	Name      string   // canonical key in Values
	Short     string   // e.g. "i" (no dash); empty = no short form
	Long      string   // e.g. "image" (no --); empty = no long form
	LongAlias []string // extra long spellings (e.g. "output" for "ouput")
	Kind      Kind
}

// Values is the parsed result keyed by canonical option name.
type Values map[string]string

// Has reports whether the flag was present.
func (v Values) Has(name string) bool {
	_, ok := v[name]
	return ok
}

// Parse parses args. It returns an error whose message matches the Ruby
// OptionParser wording where one exists.
func Parse(args []string, opts []Option) (Values, error) {
	vals := Values{}
	for i := 0; i < len(args); i++ {
		a := args[i]
		var opt *Option
		var name, token, val string
		var hasVal bool

		switch {
		case strings.HasPrefix(a, "--"):
			body := a[2:]
			if idx := strings.Index(body, "="); idx >= 0 {
				val, hasVal, body = body[idx+1:], true, body[:idx]
			}
			token = "--" + body
			opt = findLong(opts, body)
			if opt == nil {
				return nil, fmt.Errorf("invalid option: --%s", body)
			}
			name = opt.Name
		case strings.HasPrefix(a, "-") && len(a) > 1:
			body := a[1:]
			short := body[:1]
			token = "-" + short
			opt = findShort(opts, short)
			if opt == nil {
				return nil, fmt.Errorf("invalid option: -%s", short)
			}
			name = opt.Name
			if len(body) > 1 && opt.Kind != KindBool {
				val, hasVal = body[1:], true
			}
		default:
			return nil, fmt.Errorf("invalid option: %s", a)
		}

		if opt.Kind == KindBool {
			vals[name] = "true"
			continue
		}
		if !hasVal {
			if i+1 >= len(args) {
				return nil, fmt.Errorf("missing argument: %s", token)
			}
			i++
			val = args[i]
		}
		if opt.Kind == KindInt {
			if _, err := strconv.Atoi(val); err != nil {
				return nil, fmt.Errorf("invalid argument: %s %s", token, val)
			}
		}
		vals[name] = val
	}
	return vals, nil
}

func findLong(opts []Option, name string) *Option {
	for i := range opts {
		o := &opts[i]
		if o.Long == name {
			return o
		}
		for _, al := range o.LongAlias {
			if al == name {
				return o
			}
		}
	}
	return nil
}

func findShort(opts []Option, short string) *Option {
	for i := range opts {
		if opts[i].Short == short {
			return &opts[i]
		}
	}
	return nil
}

#include <stdlib.h>

int g2a_open(g2a *c, const char *path) {
    memset(c, 0, sizeof(*c));
    c->video_stream = -1;
    c->audio_stream = -1;
    if (avformat_open_input(&c->fmt, path, NULL, NULL) < 0) return -1;
    if (avformat_find_stream_info(c->fmt, NULL) < 0) return -2;
    for (unsigned i = 0; i < c->fmt->nb_streams; i++) {
        AVStream *st = c->fmt->streams[i];
        if (st->codecpar->codec_type == AVMEDIA_TYPE_VIDEO && c->video_stream < 0)
            c->video_stream = (int)i;
        if (st->codecpar->codec_type == AVMEDIA_TYPE_AUDIO && c->audio_stream < 0)
            c->audio_stream = (int)i;
    }
    if (c->video_stream >= 0) {
        AVStream *st = c->fmt->streams[c->video_stream];
        AVCodecParameters *par = st->codecpar;
        const AVCodec *codec = avcodec_find_decoder(par->codec_id);
        if (!codec) return -3;
        c->video_ctx = avcodec_alloc_context3(codec);
        if (!c->video_ctx) return -4;
        if (avcodec_parameters_to_context(c->video_ctx, par) < 0) return -5;
        if (avcodec_open2(c->video_ctx, codec, NULL) < 0) return -6;
        c->frame = av_frame_alloc();
        c->pkt = av_packet_alloc();
        c->time_base = av_q2d(st->time_base);
        c->sws = sws_getContext(c->video_ctx->width, c->video_ctx->height,
                                c->video_ctx->pix_fmt,
                                c->video_ctx->width, c->video_ctx->height,
                                AV_PIX_FMT_RGBA, SWS_BILINEAR, NULL, NULL, NULL);
        if (!c->sws) return -20;
    }
    if (c->audio_stream >= 0) {
        AVStream *st = c->fmt->streams[c->audio_stream];
        AVCodecParameters *par = st->codecpar;
        const AVCodec *codec = avcodec_find_decoder(par->codec_id);
        if (!codec) return -7;
        c->audio_ctx = avcodec_alloc_context3(codec);
        if (!c->audio_ctx) return -8;
        if (avcodec_parameters_to_context(c->audio_ctx, par) < 0) return -9;
        if (avcodec_open2(c->audio_ctx, codec, NULL) < 0) return -10;
        c->aframe = av_frame_alloc();
        c->nb_channels = par->ch_layout.nb_channels;
        c->sample_rate = par->sample_rate;
        AVChannelLayout out_layout;
        av_channel_layout_default(&out_layout, c->nb_channels);
        if (swr_alloc_set_opts2(&c->swr, &out_layout, AV_SAMPLE_FMT_S16,
                                c->sample_rate, &par->ch_layout,
                                (enum AVSampleFormat)par->format, c->sample_rate,
                                0, NULL) < 0) return -11;
        if (swr_init(c->swr) < 0) return -12;
    }
    return 0;
}

int g2a_next_frame(g2a *c) {
    if (c->video_stream < 0) return 0;
    for (;;) {
        int ret = av_read_frame(c->fmt, c->pkt);
        if (ret < 0) {
            avcodec_send_packet(c->video_ctx, NULL);
            if (avcodec_receive_frame(c->video_ctx, c->frame) >= 0) return 1;
            return 0;
        }
        if (c->pkt->stream_index == c->video_stream) {
            if (avcodec_send_packet(c->video_ctx, c->pkt) == 0) {
                if (avcodec_receive_frame(c->video_ctx, c->frame) == 0) {
                    av_packet_unref(c->pkt);
                    return 1;
                }
            }
        }
        av_packet_unref(c->pkt);
    }
}

void g2a_frame_time(g2a *c, double *t) {
    if (!c->first_pts_set) {
        c->first_pts = (c->frame->pts != AV_NOPTS_VALUE)
                           ? c->frame->pts * c->time_base
                           : 0.0;
        c->first_pts_set = 1;
    }
    double frm = (c->frame->pts != AV_NOPTS_VALUE)
                     ? c->frame->pts * c->time_base
                     : c->first_pts;
    *t = frm - c->first_pts;
}

uint8_t *g2a_export_frame(g2a *c, int *w, int *h) {
    // The sws context was built for video_ctx dims; if the decoded frame
    // differs (mid-stream resolution change / odd cropped stream), bail out
    // rather than risk an undersized sws_scale destination.
    if (c->frame->width != c->video_ctx->width ||
        c->frame->height != c->video_ctx->height)
        return NULL;
    *w = c->frame->width;
    *h = c->frame->height;
    int size = (*w) * (*h) * 4;
    uint8_t *rgba = (uint8_t *)av_malloc((size_t)size);
    if (!rgba) return NULL;
    uint8_t *dst[4] = {rgba, NULL, NULL, NULL};
    int dst_stride[4] = {(*w) * 4, 0, 0, 0};
    sws_scale(c->sws, (const uint8_t *const *)c->frame->data,
              c->frame->linesize, 0, c->frame->height, dst, dst_stride);
    return rgba;
}

static int g2a_audio_convert(g2a *c) {
    uint8_t *out[1] = {NULL};
    int out_linesize = 0;
    int out_samples = swr_get_out_samples(c->swr, c->aframe->nb_samples) + 256;
    if (av_samples_alloc(out, &out_linesize, c->nb_channels, out_samples,
                         AV_SAMPLE_FMT_S16, 0) < 0)
        return -1;
    int converted = swr_convert(c->swr, out, out_samples,
                                (const uint8_t **)c->aframe->extended_data,
                                c->aframe->nb_samples);
    int bytes = converted * c->nb_channels * 2;
    if (bytes > 0) {
        if (c->audio_buf_len + bytes > c->audio_buf_cap) {
            c->audio_buf_cap = (c->audio_buf_cap + bytes) * 2 + 1024;
            c->audio_buf = (uint8_t *)realloc(c->audio_buf, c->audio_buf_cap);
            if (!c->audio_buf) { av_freep(&out[0]); return -2; }
        }
        memcpy(c->audio_buf + c->audio_buf_len, out[0], (size_t)bytes);
        c->audio_buf_len += bytes;
    }
    av_freep(&out[0]);
    return bytes;
}

int g2a_audio_next(g2a *c) {
    if (c->audio_stream < 0) return 0;
    for (;;) {
        int ret = av_read_frame(c->fmt, c->pkt);
        if (ret < 0) {
            avcodec_send_packet(c->audio_ctx, NULL);
            while (avcodec_receive_frame(c->audio_ctx, c->aframe) == 0)
                g2a_audio_convert(c);
            return 0;
        }
        if (c->pkt->stream_index == c->audio_stream) {
            if (avcodec_send_packet(c->audio_ctx, c->pkt) == 0) {
                int produced = 0;
                while (avcodec_receive_frame(c->audio_ctx, c->aframe) == 0) {
                    g2a_audio_convert(c);
                    produced = 1;
                }
                av_packet_unref(c->pkt);
                if (produced) return 1;
                continue;
            }
        }
        av_packet_unref(c->pkt);
    }
}

void g2a_audio_take(g2a *c, uint8_t **buf, int *len) {
    *buf = c->audio_buf;
    *len = c->audio_buf_len;
    c->audio_buf = NULL;
    c->audio_buf_cap = 0;
    c->audio_buf_len = 0;
}

double g2a_duration(g2a *c) {
    if (c->fmt && c->fmt->duration != AV_NOPTS_VALUE)
        return (double)c->fmt->duration / AV_TIME_BASE;
    if (c->video_stream >= 0) {
        AVStream *st = c->fmt->streams[c->video_stream];
        if (st->duration != AV_NOPTS_VALUE) return st->duration * av_q2d(st->time_base);
    }
    return 0.0;
}

void g2a_free(void *p) {
    if (p) av_free(p);
}

void g2a_close(g2a *c) {
    if (c->sws) sws_freeContext(c->sws);
    if (c->swr) swr_free(&c->swr);
    if (c->frame) av_frame_free(&c->frame);
    if (c->aframe) av_frame_free(&c->aframe);
    if (c->pkt) av_packet_free(&c->pkt);
    if (c->video_ctx) avcodec_free_context(&c->video_ctx);
    if (c->audio_ctx) avcodec_free_context(&c->audio_ctx);
    if (c->fmt) avformat_close_input(&c->fmt);
    if (c->audio_buf) {
        free(c->audio_buf);
        c->audio_buf = NULL;
    }
}

//go:build cgo

// Package ffmpeg is a thin cgo binding over FFmpeg for frame decoding and
// audio extraction. It requires FFmpeg >= 6 and CGO_ENABLED=1.
package ffmpeg

/*
#cgo pkg-config: libavformat libavcodec libavutil libswscale libswresample
#cgo CFLAGS: -Wno-deprecated-declarations
#include <stdlib.h>
#include "ffmpeg.h"
*/
import "C"

import (
	"fmt"
	"math"
	"unsafe"
)

// Frame is a decoded RGBA image.
type Frame struct {
	Index  int
	Width  int
	Height int
	Pixels []byte // RGBA, len == Width*Height*4
}

// AudioData is interleaved signed 16-bit PCM.
type AudioData struct {
	PCM        []byte
	SampleRate int
	Channels   int
}

// DecodeFrames decodes the video at path, yielding one frame per
// stepDuration seconds (relative to the first decoded frame's timestamp).
// yield is called sequentially by the decoder goroutine.
func DecodeFrames(path string, stepDuration float64, yield func(Frame) error) error {
	cpath := C.CString(path)
	defer C.free(unsafe.Pointer(cpath))

	var c C.g2a
	if rc := C.g2a_open(&c, cpath); rc < 0 {
		return fmt.Errorf("open %s: ffmpeg error %d", path, int(rc))
	}
	defer C.g2a_close(&c)
	if c.video_stream < 0 {
		return fmt.Errorf("no video stream in %s", path)
	}

	lastK := -1
	index := 0
	for {
		rc := C.g2a_next_frame(&c)
		if rc < 0 {
			return fmt.Errorf("decode: ffmpeg error %d", int(rc))
		}
		if rc == 0 {
			return nil // EOF
		}
		var t C.double
		C.g2a_frame_time(&c, &t)
		k := int(math.Floor(float64(t)/stepDuration + 0.5))
		if k <= lastK {
			continue
		}
		lastK = k

		var w, h C.int
		buf := C.g2a_export_frame(&c, &w, &h)
		if buf == nil {
			return fmt.Errorf("decode: export frame failed")
		}
		n := int(w) * int(h) * 4
		pixels := C.GoBytes(unsafe.Pointer(buf), C.int(n))
		C.g2a_free(unsafe.Pointer(buf))

		if err := yield(Frame{Index: index, Width: int(w), Height: int(h), Pixels: pixels}); err != nil {
			return err
		}
		index++
	}
}

// DecodeAudio decodes the audio stream at path into interleaved S16 PCM.
// Returns nil, nil if the file has no audio stream.
func DecodeAudio(path string) (*AudioData, error) {
	cpath := C.CString(path)
	defer C.free(unsafe.Pointer(cpath))

	var c C.g2a
	if rc := C.g2a_open(&c, cpath); rc < 0 {
		return nil, fmt.Errorf("open %s: ffmpeg error %d", path, int(rc))
	}
	defer C.g2a_close(&c)
	if c.audio_stream < 0 {
		return nil, nil
	}
	for {
		rc := C.g2a_audio_next(&c)
		if rc < 0 {
			return nil, fmt.Errorf("audio decode: ffmpeg error %d", int(rc))
		}
		if rc == 0 {
			break
		}
	}
	var buf *C.uint8_t
	var n C.int
	C.g2a_audio_take(&c, &buf, &n)
	if buf == nil || n == 0 {
		return nil, nil
	}
	pcm := C.GoBytes(unsafe.Pointer(buf), n)
	C.free(unsafe.Pointer(buf))
	return &AudioData{PCM: pcm, SampleRate: int(c.sample_rate), Channels: int(c.nb_channels)}, nil
}

// Duration returns the video length in seconds.
func Duration(path string) (float64, error) {
	cpath := C.CString(path)
	defer C.free(unsafe.Pointer(cpath))

	var c C.g2a
	if rc := C.g2a_open(&c, cpath); rc < 0 {
		return 0, fmt.Errorf("open %s: ffmpeg error %d", path, int(rc))
	}
	defer C.g2a_close(&c)
	return float64(C.g2a_duration(&c)), nil
}

// Package audio handles WAV encoding and playback.
package audio

import (
	"bytes"
	"encoding/binary"
	"errors"
)

// EncodeWAV wraps 16-bit interleaved PCM samples in a standard 44-byte
// WAV header (PCM, mono/stereo+).
func EncodeWAV(sampleRate, channels int, pcm []byte) ([]byte, error) {
	if sampleRate <= 0 || channels <= 0 {
		return nil, errors.New("wav: invalid sample rate or channels")
	}
	if len(pcm)%2 != 0 {
		return nil, errors.New("wav: pcm must be byte-aligned to 16-bit samples")
	}
	dataLen := uint32(len(pcm))
	byteRate := uint32(sampleRate * channels * 2)
	blockAlign := uint16(channels * 2)

	buf := new(bytes.Buffer)
	buf.WriteString("RIFF")
	binary.Write(buf, binary.LittleEndian, dataLen+36)
	buf.WriteString("WAVE")
	buf.WriteString("fmt ")
	binary.Write(buf, binary.LittleEndian, uint32(16))
	binary.Write(buf, binary.LittleEndian, uint16(1)) // PCM
	binary.Write(buf, binary.LittleEndian, uint16(channels))
	binary.Write(buf, binary.LittleEndian, uint32(sampleRate))
	binary.Write(buf, binary.LittleEndian, byteRate)
	binary.Write(buf, binary.LittleEndian, blockAlign)
	binary.Write(buf, binary.LittleEndian, uint16(16)) // bits per sample
	buf.WriteString("data")
	binary.Write(buf, binary.LittleEndian, dataLen)
	buf.Write(pcm)
	return buf.Bytes(), nil
}

// Package tasker provides a goroutine worker pool that renders frames in
// parallel and reports a Ruby-style progress line.
package tasker

import (
	"fmt"
	"os"
	"sync"
	"time"
)

// Run processes indexes 0..total-1 with n workers, calling fn for each.
// fn may be called concurrently. The first error is returned; remaining
// items still run. Progress is printed to stdout in the Ruby format:
//
//	processing...  xx.xx % (time: xx.xx s)
//
// If total == 0, the percentage is omitted.
func Run(n, total int, fn func(i int) error) error {
	if n < 1 {
		n = 1
	}
	ch := make(chan int, n)
	go func() {
		defer close(ch)
		for i := 0; i < total; i++ {
			ch <- i
		}
	}()
	return RunParallel(n, total, ch, fn)
}

// RunParallel processes items from ch with n workers. Unlike Run, the item
// channel is closed by the producer; progress total is advisory (used for the
// percentage; total <= 0 shows a frame count instead). fn may be concurrent.
func RunParallel[T any](n int, total int, ch <-chan T, fn func(T) error) error {
	if n < 1 {
		n = 1
	}
	start := time.Now()
	var wg sync.WaitGroup
	var mu sync.Mutex
	var done int
	var firstErr error
	var errMu sync.Mutex
	report := func() {
		if total <= 0 {
			fmt.Printf("\rprocessing...  frames: %d (time: %.2f s)", done, time.Since(start).Seconds())
		} else {
			fmt.Printf("\rprocessing...  %.2f %% (time: %.2f s)",
				100.0*float64(done)/float64(total), time.Since(start).Seconds())
		}
	}
	for w := 0; w < n; w++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			for item := range ch {
				if err := fn(item); err != nil {
					errMu.Lock()
					if firstErr == nil {
						firstErr = err
					}
					errMu.Unlock()
					continue
				}
				mu.Lock()
				done++
				report()
				mu.Unlock()
			}
		}()
	}
	wg.Wait()
	fmt.Fprint(os.Stdout, "\r\x1b[K")
	return firstErr
}
```

import (
	"encoding/json"
	"fmt"
	"image"
	"os"

	"sort"
	"strconv"

	"github.com/Mark24Code/convert2ascii/go2ascii/internal/tasker"
)

// DefaultStepDuration mirrors the Ruby DEFAULT_STEP_DURATION (≈25 fps).
const DefaultStepDuration = 0.04

// VideoOptions mirrors the Ruby Video2Ascii constructor kwargs.
type VideoOptions struct {
	URI         string
	Width       int
	Style       string
	Color       string
	ColorBlock  bool
	StepDuration float64
}

// Video2Ascii extracts a video's frames as ASCII and optionally saves/plays them.
type Video2Ascii struct {
	URI          string
	Width        int
	Style        string
	Color        string
	ColorBlock   bool
	StepDuration float64
	Threads      int

	tmpdir     string
	framePaths []string
	audioPath  string
}

// NewVideo2Ascii builds a Video2Ascii. Width 0 means terminal columns.
func NewVideo2Ascii(opts VideoOptions) *Video2Ascii {
	width := opts.Width
	if width <= 0 {
		_, cols := ansi.Size()
		width = cols
	}
	step := opts.StepDuration
	if step <= 0 {
		step = DefaultStepDuration
	}
	return &Video2Ascii{
		URI:          opts.URI,
		Width:        width,
		Style:        opts.Style,
		Color:        opts.Color,
		ColorBlock:   opts.ColorBlock,
		StepDuration: step,
		Threads:      runtime.NumCPU(),
	}
}

// Generate extracts audio + frames into ~/.convert2ascii and converts every
// frame to N.txt in parallel. It mutates and returns self on success.
func (v *Video2Ascii) Generate() error {
	home, err := os.UserHomeDir()
	if err != nil {
		return err
	}
	v.tmpdir = filepath.Join(home, ".convert2ascii")
	if err := os.RemoveAll(v.tmpdir); err != nil {
		return err
	}
	if err := os.MkdirAll(v.tmpdir, 0o755); err != nil {
		return err
	}
	// Reset state so a second Generate on the same instance re-extracts audio
	// instead of skipping it after RemoveAll wiped the previous audio.wav.
	v.audioPath = ""
	v.framePaths = nil

	fmt.Println(green("[info] parsing audio..."))
	ad, err := ffmpeg.DecodeAudio(v.URI)
	if err != nil {
		return err
	}
	if ad != nil {
		wavData, err := audio.EncodeWAV(ad.SampleRate, ad.Channels, ad.PCM)
		if err != nil {
			return err
		}
		v.audioPath = filepath.Join(v.tmpdir, "audio.wav")
		if err := os.WriteFile(v.audioPath, wavData, 0o644); err != nil {
			return err
		}
	}
	fmt.Println(green("[info] parsing audio done."))

	fmt.Println(green("[info] video slicing..."))

	total, err := frameCount(v.URI, v.StepDuration)
	if err != nil {
		return err
	}

	frameCh := make(chan ffmpeg.Frame, v.Threads*2)
	var decErr error
	go func() {
		defer close(frameCh)
		decErr = ffmpeg.DecodeFrames(v.URI, v.StepDuration, func(fr ffmpeg.Frame) error {
			frameCh <- fr
			return nil
		})
	}()

	renderOpts := imagelib.Options{
		Width:      v.Width,
		Style:      imagelib.Style(v.Style),
		Color:      imagelib.ColorMode(v.Color),
		ColorBlock: v.ColorBlock,
	}

	convErr := tasker.RunParallel(v.Threads, total, frameCh, func(fr ffmpeg.Frame) error {
		rgba := &image.RGBA{
			Pix:    fr.Pixels,
			Stride: fr.Width * 4,
			Rect:   image.Rect(0, 0, fr.Width, fr.Height),
		}
		s := imagelib.Render(rgba, renderOpts)
		// 1-based frame numbers match the Ruby output (1.txt, 2.txt, ...).
		name := strconv.Itoa(fr.Index+1) + ".txt"
		return os.WriteFile(filepath.Join(v.tmpdir, name), []byte(s+"\n"), 0o644)
	})
	if decErr != nil {
		return fmt.Errorf("decode frames: %w", decErr)
	}
	if convErr != nil {
		return fmt.Errorf("convert frames: %w", convErr)
	}

	v.framePaths, err = filepath.Glob(filepath.Join(v.tmpdir, "*.txt"))
	if err != nil {
		return err
	}
	sort.Slice(v.framePaths, func(i, j int) bool {
		return frameOrder(v.framePaths[i]) < frameOrder(v.framePaths[j])
	})

	meta := map[string]any{
		"step_duration": v.StepDuration,
		"audio":         nil,
		"frames_count":  len(v.framePaths),
	}
	if v.audioPath != "" {
		meta["audio"] = filepath.Base(v.audioPath)
	}
	metaRaw, err := json.MarshalIndent(meta, "", "  ")
	if err != nil {
		return err
	}
	if err := os.WriteFile(filepath.Join(v.tmpdir, "meta.json"), metaRaw, 0o644); err != nil {
		return err
	}

	fmt.Println(green("[info] video slicing done."))
	return nil
}

// Save copies the generated frames + audio into outputDir.
func (v *Video2Ascii) Save(outputDir string) error {
	if v.tmpdir == "" {
		return fmt.Errorf("nothing to save: call Generate first")
	}
	if err := os.RemoveAll(outputDir); err != nil {
		return err
	}
	if err := os.MkdirAll(outputDir, 0o755); err != nil {
		return err
	}
	entries, err := os.ReadDir(v.tmpdir)
	if err != nil {
		return err
	}
	for _, e := range entries {
		if strings.HasSuffix(e.Name(), ".jpg") {
			continue
		}
		src := filepath.Join(v.tmpdir, e.Name())
		dst := filepath.Join(outputDir, e.Name())
		if e.IsDir() {
			continue
		}
		data, err := os.ReadFile(src)
		if err != nil {
			return err
		}
		if err := os.WriteFile(dst, data, 0o644); err != nil {
			return err
		}
	}
	fmt.Println(green("[info] save success!"))
	return nil
}

// AfterClean removes the staging directory ~/.convert2ascii.
func (v *Video2Ascii) AfterClean() {
	if v.tmpdir != "" {
		_ = os.RemoveAll(v.tmpdir)
		v.tmpdir = ""
	}
}

func frameCount(uri string, step float64) (int, error) {
	d, err := ffmpeg.Duration(uri)
	if err != nil {
		return 0, err
	}
	if d <= 0 {
		return 0, nil
	}
	return int(d/step) + 1, nil
}

func frameOrder(p string) int {
	n, _ := strconv.Atoi(strings.TrimSuffix(filepath.Base(p), ".txt"))
	return n
}

func green(s string) string {
	return "\x1b[32m" + s + "\x1b[0m"
}

	"github.com/gopxl/beep/v2/mp3"
	"github.com/gopxl/beep/v2/wav"
	"github.com/hajimehoshi/oto/v2"
)

// Player plays a local audio file (wav or mp3) and exposes a playback clock
// in seconds. Position() is derived from a sample counter, giving a stable
// master clock for A/V sync.
type Player struct {
	ctx     *oto.Context
	out     *oto.Player
	counter *sampleCounter
	once    sync.Once
}

type sampleCounter struct {
	beep.Streamer
	mu   sync.Mutex
	pos  int64
	rate float64
}

func (c *sampleCounter) Stream(samples [][2]float64) (int, bool) {
	n, ok := c.Streamer.Stream(samples)
	c.mu.Lock()
	c.pos += int64(n)
	c.mu.Unlock()
	return n, ok
}

func (c *sampleCounter) position() float64 {
	c.mu.Lock()
	defer c.mu.Unlock()
	return float64(c.pos) / c.rate
}

// NewPlayer opens path (wav or mp3) and prepares an oto player. If loop is
// true, the audio repeats indefinitely.
func NewPlayer(path string, loop bool) (*Player, error) {
	f, err := os.Open(path)
	if err != nil {
		return nil, err
	}
	defer f.Close()

	streamer, format, err := decode(f, path)
	if err != nil {
		return nil, err
	}

	buffered := beep.NewBuffer(format)
	buffered.Store(streamer)
	if err := streamer.Close(); err != nil {
		return nil, err
	}

	var src beep.Streamer = buffered.Streamer(0, buffered.Len())
	if loop {
		src = beep.Loop(-1, src)
	}

	ctx, err := oto.NewContext(format.SampleRate, format.NumChannels, format.Precision)
	if err != nil {
		return nil, err
	}
	// The counter MUST be the oto source (oto pulls samples from it), so the
	// sample clock actually advances; wrapping src without wiring it in would
	// leave Position() stuck at 0 forever.
	counter := &sampleCounter{Streamer: src, rate: float64(format.SampleRate)}
	out := ctx.NewPlayer(counter)
	return &Player{
		ctx:     ctx,
		out:     out,
		counter: counter,
	}, nil
}

// Play starts audio output.
func (p *Player) Play() {
	if p.out != nil {
		p.out.Play()
	}
}

// Position returns seconds of audio played so far.
func (p *Player) Position() float64 {
	if p.counter == nil {
		return 0
	}
	return p.counter.position()
}

// Close stops audio and releases resources.
func (p *Player) Close() error {
	var err error
	p.once.Do(func() {
		if p.out != nil {
			p.out.Close()
		}
		if p.ctx != nil {
			p.ctx.Close()
		}
	})
	return err
}

func decode(f *os.File, path string) (beep.StreamSeekCloser, beep.Format, error) {
	switch strings.ToLower(filepath.Ext(path)) {
	case ".mp3":
		return mp3.Decode(f)
	default:
		return wav.Decode(f)
	}
}

```go
// Package player renders a sequence of ASCII frames in the terminal with
// self-adaptive A/V sync, mirroring the Ruby TerminalPlayer.
package player

import (
	"errors"
	"fmt"
	"os"
	"os/signal"
	"strings"
	"time"

	"github.com/Mark24Code/convert2ascii/go2ascii/internal/ansi"
	"github.com/Mark24Code/convert2ascii/go2ascii/internal/audio"
)

// SAFE_SLOW_DELTA / SAFE_FAST_DELTA mirror the Ruby tolerances.
const (
	SAFE_SLOW_DELTA = 0.9 // seconds: video lags audio beyond this -> skip frames
	SAFE_FAST_DELTA = 0.2 // seconds: video leads audio beyond this -> pause
)

// AudioClock is the minimal interface the player needs from the audio layer,
// so it can run without a sound device.
type AudioClock interface {
	Play()
	Position() float64
	Close() error
}

// wallClock is the wall-clock fallback for videos without audio.
func wallClock(start time.Time) float64 { return time.Since(start).Seconds() }

// Options configure a playback session.
type Options struct {
	Frames       []string
	AudioPath    string
	StepDuration float64
	PlayLoop     bool
	Debug        bool
}

// Player plays frames with optional audio.
type Player struct {
	opts Options
}

// New builds a Player.
func New(o Options) *Player { return &Player{opts: o} }

// Play runs the playback loop. It blocks until playback ends or is interrupted.
func (p *Player) Play() error {
	o := p.opts
	if len(o.Frames) == 0 {
		return errors.New("[Error] frame's length must be >= 0")
	}
	rows, _ := ansi.Size()

	var clock AudioClock
	if o.AudioPath != "" {
		ap, err := audio.NewPlayer(o.AudioPath, o.PlayLoop)
		if err != nil {
			// Ruby degrades to silent wall-clock playback when ffplay fails;
			// mirror that: warn and play without audio.
			fmt.Fprintf(os.Stderr, "[warn] audio unavailable, playing without audio: %v\n", err)
		} else {
			clock = ap
		}
	}

	// Ctrl-C: clean up and exit 0, like the Ruby trap. The done channel lets
	// the handler goroutine exit on the normal return path (signal.Stop does
	// not unblock a plain <-sig, so without it every Play leaks a goroutine).
	sig := make(chan os.Signal, 1)
	signal.Notify(sig, os.Interrupt)
	defer signal.Stop(sig)
	done := make(chan struct{})
	defer close(done)
	go func() {
		select {
		case <-sig:
			cleanup(clock)
			os.Exit(0)
		case <-done:
		}
	}()

	ansi.OpenBuffer()
	ansi.HideCursor()
	ansi.ClearScreen()
	defer cleanup(clock)

	if clock != nil {
		clock.Play()
	}

	step := o.StepDuration
	if step <= 0 {
		step = 0.04
	}
	start := time.Now()
	frameIndex := 0
	firstFrame := true
	// Backspace by the terminal height (fullScreen pads to `rows`), not
	// len(frames) — Ruby's len(frames)+1 is a latent bug that misrenders
	// clips shorter than the terminal.
	backspace := strings.Repeat("\x1b[A", rows+1)
	totalDuration := float64(len(o.Frames)) * step

	for {
		content := o.Frames[frameIndex]
		if !firstFrame {
			fmt.Print(backspace)
		}
		ansi.ClearScreen()
		fmt.Print(fullScreen(content, rows))
		firstFrame = false
		time.Sleep(time.Duration(step * float64(time.Second)))

		var actual float64
		if clock != nil {
			actual = clock.Position()
		} else {
			actual = wallClock(start)
		}
		videoTime := float64(frameIndex) * step
		offset := actual - videoTime
		advance := syncOffset(offset, step, SAFE_SLOW_DELTA, SAFE_FAST_DELTA)
		frameIndex = (frameIndex + advance) % len(o.Frames)

		if !o.PlayLoop {
			if time.Since(start).Seconds() > totalDuration {
				break
			}
		}
		if o.Debug {
			fmt.Printf("\nRealTime: %.2f s\nFrameTime: %.2f s\nCurrentFrame: %d / %d\nOffset: %.2f s\nAdvance: %d\n",
				time.Since(start).Seconds(), float64(frameIndex+1)*step, frameIndex, len(o.Frames), offset, advance)
		}
	}
	return nil
}

// syncOffset computes how many frames to advance given the offset between the
// audio clock and the video clock. Mirrors the Ruby self_adaption_frame_play.
func syncOffset(offset float64, step, slowDelta, fastDelta float64) int {
	switch {
	case offset > slowDelta:
		return int(offset / step)
	case offset < -fastDelta:
		return 0
	default:
		return 1
	}
}

// fullScreen pads content to the terminal height (or truncates it).
func fullScreen(content string, rows int) string {
	lines := strings.Split(content, "\n")
	if len(lines) > rows {
		return strings.Join(lines[:rows], "\n")
	}
	for len(lines) < rows {
		lines = append(lines, "")
	}
	return strings.Join(lines, "\n")
}

func cleanup(clock AudioClock) {
	ansi.ShowCursor()
	ansi.ClearScreen()
	ansi.CloseBuffer()
	if clock != nil {
		_ = clock.Close()
	}
}
```

> Note: `wallClock(start)` is the no-audio fallback clock; with audio, `AudioClock.Position()` is the master clock.

// Player replays saved ASCII frames, mirroring Convert2Ascii::TerminalPlayer.
type Player struct {
	Frames       []string
	AudioPath    string
	StepDuration float64
	PlayLoop     bool
	Debug        bool
}

// Play runs the playback until it ends or is interrupted.
func (p *Player) Play() error {
	return player.New(player.Options{
		Frames:       p.Frames,
		AudioPath:    p.AudioPath,
		StepDuration: p.StepDuration,
		PlayLoop:     p.PlayLoop,
		Debug:        p.Debug,
	}).Play()
}

type metaConfig struct {
	StepDuration float64 `json:"step_duration"`
	Audio        *string `json:"audio"`
	FramesCount  int     `json:"frames_count"`
}

// PlayFrames replays a saved frames directory (N.txt + meta.json [+ audio]),
// mirroring the Ruby exe's play_frames helper.
func PlayFrames(dir string, playLoop bool) error {
	raw, err := os.ReadFile(filepath.Join(dir, "meta.json"))
	if err != nil {
		return fmt.Errorf("read meta.json: %w", err)
	}
	var cfg metaConfig
	if err := json.Unmarshal(raw, &cfg); err != nil {
		return fmt.Errorf("parse meta.json: %w", err)
	}

	txts, err := filepath.Glob(filepath.Join(dir, "*.txt"))
	if err != nil {
		return err
	}
	sort.Slice(txts, func(i, j int) bool {
		ni, _ := strconv.Atoi(strings.TrimSuffix(filepath.Base(txts[i]), ".txt"))
		nj, _ := strconv.Atoi(strings.TrimSuffix(filepath.Base(txts[j]), ".txt"))
		return ni < nj
	})
	frames := make([]string, 0, len(txts))
	for _, fp := range txts {
		b, err := os.ReadFile(fp)
		if err != nil {
			return err
		}
		frames = append(frames, string(b))
	}

	audioPath := ""
	if cfg.Audio != nil && *cfg.Audio != "" {
		audioPath = filepath.Join(dir, *cfg.Audio)
	}

	p := &Player{
		Frames:       frames,
		AudioPath:    audioPath,
		StepDuration: cfg.StepDuration,
		PlayLoop:     playLoop,
	}
	return p.Play()
}

// Play plays the generated frames with audio. It is defined here (not in
// video2ascii.go) because it depends on internal/player, which only exists
// from Task 12 onward. Returns after playback ends.
func (v *Video2Ascii) Play(playLoop bool) error {
	if v.tmpdir == "" {
		return fmt.Errorf("nothing to play: call Generate first")
	}
	frames := make([]string, 0, len(v.framePaths))
	for _, fp := range v.framePaths {
		b, err := os.ReadFile(fp)
		if err != nil {
			return err
		}
		frames = append(frames, string(b))
	}
	p := &Player{
		Frames:       frames,
		AudioPath:    v.audioPath,
		StepDuration: v.StepDuration,
		PlayLoop:     playLoop,
	}
	defer v.AfterClean()
	return p.Play()
}

	{Name: "loop", Long: "loop", Kind: cli.KindBool},
}

const videoUsage = `Usage: video2ascii [options]

* default will generate and play without save.
* -p will just play ascii frames dir, and ignore -i, -o others options. --loop will play loop
* -i,-o will just generate and output frames and ignore others options
        --version                    version
    -i, --input=URI                  video uri (required)
    -w, --width=WIDTH                video width (integer)
    -s, --style=STYLE                ascii style: ['color'| 'text']
    -b, --block                      ascii color style use BLOCK or not [ true | false ]
    -o, --ouput=OUTPUT               save ascii frames to the output directory
    -p, --play_dir=PLAY_DIRNAME      input the ascii frames directory to play
        --loop

func parseVideoArgs(args []string) (cli.Values, error, bool) {
	vals, err := cli.Parse(args, videoOpts)
	if err != nil {
		return nil, err, false
	}
	// --help/--version must work without an input/play_dir (Ruby parses them
	// during option processing, before the mode checks).
	if vals.Has("help") || vals.Has("version") {
		return vals, nil, true
	}
	if _, ok := vals["play_dir"]; ok {
		return vals, nil, true
	}
	if _, ok := vals["input"]; !ok {
		fmt.Println("Error: --input option is required.")
		return nil, nil, false
	}
	return vals, nil, true
}

func main() {
	vals, err, ok := parseVideoArgs(os.Args[1:])
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
	if !ok {
		os.Exit(1)
	}
	if vals.Has("help") {
		fmt.Print(videoUsage)
		return
	}
	if vals.Has("version") {
		fmt.Printf("convert2ascii/video2ascii: v%s\n", version.Version)
		fmt.Printf("author: %s\n", version.Author)
		fmt.Printf("mail: %s\n", version.Mail)
		fmt.Printf("project: %s\n", version.Project)
		return
	}

	if pd, ok := vals["play_dir"]; ok {
		if err := convert2ascii.PlayFrames(pd, vals.Has("loop")); err != nil {
			fmt.Fprintf(os.Stderr, "[Error] %v\n", err)
			os.Exit(1)
		}
		return
	}

	opts := convert2ascii.VideoOptions{URI: vals["input"]}
	if w, ok := vals["width"]; ok {
		n, _ := strconv.Atoi(w)
		opts.Width = n
	}
	if s, ok := vals["style"]; ok {
		if s != "color" && s != "text" {
			fmt.Println("Error: --style option must be [\"color\" | \"text\"].")
			os.Exit(1)
		}
		opts.Style = s
	}
	opts.ColorBlock = vals.Has("block")

	v := convert2ascii.NewVideo2Ascii(opts)
	if err := v.Generate(); err != nil {
		fmt.Fprintf(os.Stderr, "[Error] %v\n", err)
		os.Exit(1)
	}

	if out, ok := vals["ouput"]; ok {
		if err := v.Save(out); err != nil {
			fmt.Fprintf(os.Stderr, "[Error] %v\n", err)
			os.Exit(1)
		}
		// Ruby's save has `ensure after_clean`; don't leak ~/.convert2ascii.
		v.AfterClean()
		return
	}

	if err := v.Play(vals.Has("loop")); err != nil {
		fmt.Fprintf(os.Stderr, "[Error] %v\n", err)
		os.Exit(1)
	}
}
